Run this on the server (Linux, Python 3.8+). The token is shown once.

sudo curl -fsSL {{ base_url }}agent/lat-agent.py -o /usr/local/bin/lat-agent
sudo chmod 755 /usr/local/bin/lat-agent
printf 'LAT_URL={{ base_url }}\nLAT_TOKEN={{ new_token }}\n' | sudo install -m 600 /dev/stdin /etc/lat-agent.env
sudo tee /etc/systemd/system/lat-agent.service >/dev/null <<'EOF'
[Unit]
Description=Lean Asset Tracker agent
After=network-online.target
Wants=network-online.target
[Service]
EnvironmentFile=/etc/lat-agent.env
ExecStart=/usr/bin/python3 /usr/local/bin/lat-agent
Restart=always
RestartSec=10
DynamicUser=yes
NoNewPrivileges=yes
ProtectSystem=strict
ProtectHome=yes
[Install]
WantedBy=multi-user.target
EOF
sudo systemctl daemon-reload && sudo systemctl enable --now lat-agent
